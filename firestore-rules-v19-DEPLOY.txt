// (user app + admin panel — একই Firebase project; staff panel deleted BUILD 29)

// মূল নীতি:
//  • public ডেটা (tournaments, banners, results) সবাই পড়তে পারবে
//  • join flow-এর lock/counter লেখা user নিজে করবে — কিন্তু কঠোর সীমার ভেতর
//    (শুধু নির্দিষ্ট ফিল্ড, শুধু নিজের ডকুমেন্ট, কাউন্টার ±1)
//  • টাকা-পয়সার ডকুমেন্ট (withdrawal, wallet) user শুধু নিজেরটা দেখবে
//  • কেউ নিজে নিজে role='admin' বা role='staff' বসাতে পারবে না

// Deploy: Firebase Console → Firestore Database → Rules ট্যাবে পেস্ট করে Publish
// (অথবা কম্পিউটার থেকে: firebase deploy --only firestore:rules)

service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helpers ──
    function isSignedIn() {
      return request.auth != null;
    }
    function isAdmin() {
      return isSignedIn()
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }
    // v19 (M6 STAFF CLEANUP): isStaff() DELETED — Staff Panel deleted
    // (BUILD 29); admin login requires role=='admin'; grep finds 0
    // isStaff() uses in the user v8.6 + admin BUILD 31 trees. Staff has
    // no permissions of any kind in v19 (no writes, no reads).
    // v14 (H4): ban enforcement in rules. The app signs a banned user out
    // of the UI, but the Auth token stays valid, so SDK writes were still
    // possible. A banned profile carries isBanned:true (admin ban flow)
    // and/or status:'banned' (app.js checks both) — either one blocks.
    // NOT applied to users create (doc doesn't exist yet) or the
    // registeredUIDs signup claim path. Any get() error fails CLOSED.
    function isNotBanned() {
      return isSignedIn()
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('isBanned', false) != true
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('status', '') != 'banned';
    }
    function isOwner(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }
    // শুধু নির্দিষ্ট ফিল্ডগুলো বদলানো যাবে (বাকি সব আগের মতোই থাকতে হবে)
    function onlyKeys(keys) {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys);
    }
    // কাউন্টার এক ধাপ (±1) বা অপরিবর্তিত — seat reservation-এর জন্য যথেষ্ট,
    // কিন্তু কেউ ইচ্ছামতো সংখ্যা বসাতে পারবে না
    function stepDelta(oldV, newV) {
      return newV == oldV || newV == oldV + 1 || newV == oldV - 1;
    }
    // v12 (AUD-007): counter INTEGRITY — stepDelta alone still let any
    // signed-in user walk the counters ±1 per write and set filledPlayers
    // to anything. Now, WHEN filledPlayers / filledSlots change, they must
    // stay consistent with filledTeams (filledSlots == filledTeams;
    // filledPlayers == filledTeams × playersPerTeam). The free-join
    // transaction and staff confirm both write exactly this shape.
    // Updates that don't touch those fields (e.g. staff reject touching
    // only pendingTeams) are not re-validated against legacy data.
    // (a) reverted 2026-10-25: the shapedTeamCreate() helper lived here
    // and was deleted with the owner veto — team create is open again
    // (see the tournamentTeams rule below).
    // v18 (C4): legacy tournament doc-এ playersPerTeam field নাও থাকতে
    // পারে — তখন teamFormat থেকে default (SOLO 1 / DUO 2 / SQUAD 4)।
    // আগে hardcoded 1 ছিল, তাই DUO/SQUAD legacy join permission-denied-এ
    // মরত (code ×2/×4 লেখে, rules ×1 চাইত)।
    function defaultPlayersPerTeam() {
      let fmt = request.resource.data.get('teamFormat', 'SOLO');
      return fmt == 'DUO' ? 2 : (fmt == 'SQUAD' ? 4 : 1);
    }
    function counterGuard() {
      let changed = request.resource.data.diff(resource.data).affectedKeys();
      return (!changed.hasAny(['filledPlayers'])
              || request.resource.data.filledPlayers
                 == request.resource.data.filledTeams * request.resource.data.get('playersPerTeam', defaultPlayersPerTeam()))
          && (!changed.hasAny(['filledSlots'])
              || request.resource.data.filledSlots == request.resource.data.filledTeams);
    }

    // ── Users ──
    match /users/{uid} {
      // v11 (A1): owner/admin/staff only. আগে isSignedIn() ছিল — যেকোনো
      // logged-in user অন্যের email/ffUID/balance পড়তে পারত। User app
      // এখন শুধু নিজের doc পড়ে; leaderboard published
      // leaderboard/current doc থেকে আসে (live users fallback v7.6-এ
      // সরানো হয়েছে)।
      allow read: if isOwner(uid) || isAdmin();
      // v11 (A1): create শুধু owner; role admin/staff নয়; টাকার তিনটা
      // counter-ই 0 দিয়ে শুরু (user-app register + admin 'Create Admin
      // Account' দুটো write-ই হুবহু এটা করে — যাচাই করা)। .get(...,0)
      // ব্যবহার করা হলো যাতে field বাদ গেলেও (মানে 0) rule error না হয়,
      // কিন্তু non-zero বসালেই deny।
      allow create: if isOwner(uid)
        && (!('role' in request.resource.data)
            || (request.resource.data.role != 'admin' && request.resource.data.role != 'staff'))
        && request.resource.data.get('walletBalance', 0) == 0
        && request.resource.data.get('totalEarnings', 0) == 0
        && request.resource.data.get('totalWins', 0) == 0
        // v16 (wallet): the two new money counters also start at 0 —
        // depositBalance (entry-fee only, never withdrawable) and
        // totalDeposited (lifetime deposited, display only).
        && request.resource.data.get('depositBalance', 0) == 0
        && request.resource.data.get('totalDeposited', 0) == 0;
      // v11 (A1): owner শুধু এই ফিল্ডগুলো বদলাতে পারবে — user app-এর
      // প্রতিটা আসল write অডিট করে (account-settings name,
      // profile-photo photoURL, join-flow joinedTournaments arrayUnion,
      // updatedAt)। walletBalance/totalEarnings/totalWins/role console
      // থেকে নিজে বদলানো যাবে না। admin সব ফিল্ড।

    // registeredUIDs/{ffUID} doc বানায় { userId, registeredAt } ফিল্ডসহ
    // (v18 C5: email আর এখানে লেখা হয় না — users doc-এই আছে, আর get
    // rule খোলা থাকায় roster-এর FF UID দিয়ে email লিক হতো),
    // আগে get() করে existence চেক করে। তাই:
    //  • get (একটা doc পড়া): login করা user — existence চেকের জন্য
    //  • list (সব স্ক্যান): শুধু admin (FF UID→email ম্যাপিং লিক হবে না)
    //  • create: শুধু নিজের auth uid দিয়ে (userId == request.auth.uid)
    //  • update/delete: শুধু admin
    match /registeredUIDs/{ffUid} {
      allow get: if isSignedIn() || isAdmin();
      allow list: if isAdmin();
      // v12 (AUD-018): the claim must match the caller's OWN profile doc —
      // the register flow writes users/{uid} (with ffUID) FIRST, then claims.
      // Before, any signed-in user could claim ANY FF UID directly and
      // squat other players' UIDs before they signed up.
      allow create: if isSignedIn()
        && request.resource.data.userId == request.auth.uid
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('ffUID', '') == ffUid;
      // v8.11 (B4): re-claim ordering — the client updates users/{uid}.ffUID
      // FIRST, then creates the NEW claim (the unchanged create rule above
      // checks the users doc ffUID), then deletes the OLD claim. Owner may
      // delete ONLY their own claim; update stays admin-only.
      allow update: if isAdmin();
      allow delete: if isAdmin() || (isSignedIn() && resource.data.userId == request.auth.uid);
    }

    // ── Banned devices (v19 MERGE — device ban) ──
    // DESIGN: `allow get: if true` (public single-doc get) is REQUIRED,
    // not a lapse. The registration check runs BEFORE sign-in (user
    // auth.service.js checks the device before the Auth account exists;
    // admin auth.js doAdminRegister likewise), so a signed-in-only rule
    // would fail open by design — both apps return "not banned" on a
    // denied read (device-ban.js isDeviceBanned, fail open by design).
    // Device ids are unguessable random UUIDs in localStorage; `list` is
    // NEVER allowed, so ids cannot be enumerated — only a caller who
    // already knows one exact device id can probe that one doc
    // (existence oracle). RESIDUAL RISK, accepted: anyone who learns a
    // device id can test whether it is banned; and because rules cannot
    // check arrayUnion contents, an owner could plant someone else's id
    // in their own deviceIds (see users rule note) so a ban of them
    // would ban that device too. The fail-open alternative is worse.
    // Writes are admin-only (Users ban action, users.js:613 setDoc with
    // merge). Unban never deletes these docs, but admin delete is allowed
    // so a wrongly-banned device can be cleared manually.
    match /bannedDevices/{deviceId} {
      allow get: if true;
      allow list: if false;
      allow create, update, delete: if isAdmin();
    }

    // ── Payment requests (join fee) ──
    match /paymentRequests/{reqId} {
      // v12 (AUD-003): was read+create open to ANY signed-in user — every
      // player's UTR + email was enumerable, forged 'approved' docs could
      // be planted, and a victim's UTR doc ID could be squatted so their
      // real submission failed. Now: read = owner/admin/staff only; create
      // is shaped to EXACTLY what payment.service.js writes (own uid,
      // status 'pending', deterministic doc ID 'trx_' + transactionId,
      // amount == entryFee > 0, review fields null).
      // get is split from list: the payment flow checks a UTR doc's
      // EXISTENCE inside its transaction — for a missing doc resource is
      // null, and dereferencing it would deny the check and break every
      // first-time payment. Existence of a missing doc reveals nothing.
      // v19 (M6): staff get/list removed.
      allow get: if isAdmin()
        || (isSignedIn() && (resource == null || resource.data.userId == request.auth.uid));
      allow list: if isAdmin()
        || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow create: if isSignedIn()
        && isNotBanned()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.paymentMethod == 'upi'
        && request.resource.data.transactionId is string
        && request.resource.data.transactionId.size() >= 8
        && reqId == 'trx_' + request.resource.data.transactionId
        && request.resource.data.entryFee is number
        && request.resource.data.entryFee > 0
        && request.resource.data.amount == request.resource.data.entryFee
        && request.resource.data.get('reviewedBy', null) == null
        && request.resource.data.get('reviewedAt', null) == null
        && request.resource.data.get('rejectionReason', null) == null
        && request.resource.data.keys().hasOnly(['paymentId', 'tournamentId', 'tournamentName', 'teamId', 'teamName', 'userId', 'userName', 'userEmail', 'entryFee', 'amount', 'transactionId', 'paymentMethod', 'status', 'submittedAt', 'reviewedAt', 'reviewedBy', 'rejectionReason']);
      // v16 (wallet): WALLET-PAY branch — same shape as the UPI branch,
      // but paymentMethod 'wallet', deterministic doc ID 'wlt_' + teamId
      // and transactionId 'WALLET-' + teamId (no UTR involved). The user
      // panel never debits here; the admin debits depositBalance at
      // confirm time (admin-panel code + walletTransactions DEBIT entry).
      allow create: if isSignedIn()
        && isNotBanned()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.paymentMethod == 'wallet'
        && request.resource.data.transactionId == 'WALLET-' + request.resource.data.teamId
        && reqId == 'wlt_' + request.resource.data.teamId
        && request.resource.data.entryFee is number
        && request.resource.data.entryFee > 0
        && request.resource.data.amount == request.resource.data.entryFee
        && request.resource.data.get('reviewedBy', null) == null
        && request.resource.data.get('reviewedAt', null) == null
        && request.resource.data.get('rejectionReason', null) == null
        && request.resource.data.keys().hasOnly(['paymentId', 'tournamentId', 'tournamentName', 'teamId', 'teamName', 'userId', 'userName', 'userEmail', 'entryFee', 'amount', 'transactionId', 'paymentMethod', 'status', 'submittedAt', 'reviewedAt', 'reviewedBy', 'rejectionReason']);
      allow update, delete: if isAdmin();
    }
    // Transaction-ID ডুপ্লিকেট ঠেকানোর রেজিস্ট্রি
    match /usedTransactions/{txnId} {
      allow read: if isAdmin();
      allow write: if isAdmin();
    }

    // ── Withdrawals (v19: ADMIN-ONLY approve/reject — staff removed, BUILD 29) ──
    match /withdrawalRequests/{reqId} {
      // v18 (H4): resource == null guard — counter-tracked request doc
      // admin delete করলে txn.get() deny হয়ে সব ভবিষ্যৎ withdrawal
      // ভেঙে যেত (counter prune-ও ওই txn-এই হয়)।
      // v19 (M6): staff read removed.
      allow read: if isAdmin()
        || (isSignedIn() && (resource == null || resource.data.userId == request.auth.uid));
      // v11 (A1): create কঠোর — নিজের uid, status ঠিক 'pending',
      // amount সংখ্যা > 0, resolvedAt থাকবে না, আর key শুধু user app
      // আসলে যা লেখে (withdrawal.service.js অডিট করে: ffUid সহ)।
      // আগে শুধু userId মিললেই status:'approved' দিয়েও বানানো যেত।
      allow create: if isSignedIn()
        && isNotBanned()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.amount is number
        && request.resource.data.amount > 0
        // v14 (W11 rules part): the settings/withdrawal minAmount (₹1
        // default when the doc/field is missing) is now enforced in
        // rules, not only in client code.
        && request.resource.data.amount >= get(/databases/$(database)/documents/settings/withdrawal).data.get('minAmount', 1)
        && !('resolvedAt' in request.resource.data)
        && request.resource.data.keys().hasOnly(['userId', 'userName', 'ffUid', 'amount', 'method', 'phoneNumber', 'upiId', 'status', 'requestedAt']);
      // approve/reject: admin only (v19 M6 — staff removed).
      allow update, delete: if isAdmin();
    }

    // ── Deposits (v16, wallet top-up) ──
    // Mirrors withdrawalRequests create, but the money flows the other
    // way: a deposit request CREDITS the user on approve, so approve /
    // reject is ADMIN ONLY — staff is decrease-only (v14 AUD-004) and
    // must never approve a deposit. Amount is bounded by the
    // admin-configured settings/deposit minAmount/maxAmount (₹50/₹2,000
    // defaults when the fields are missing — the doc itself must exist,
    // see the v16 header note). Doc ID is deterministic 'dep_' + the
    // uppercased UTR (client-side; rules cannot uppercase, but the
    // doc ID IS bound to the transactionId field: reqId == 'dep_' +
    // transactionId — nobody can squat another user's UTR).
    match /depositRequests/{reqId} {
      // v18 (H4): resource == null guard — withdrawal rule-এর মতোই।
      // v19 (M6): staff read removed (update/delete were already admin-only).
      allow read: if isAdmin()
        || (isSignedIn() && (resource == null || resource.data.userId == request.auth.uid));
      allow create: if isSignedIn()
        && isNotBanned()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.amount is number
        && request.resource.data.amount > 0
        && request.resource.data.amount >= get(/databases/$(database)/documents/settings/deposit).data.get('minAmount', 50)
        && request.resource.data.amount <= get(/databases/$(database)/documents/settings/deposit).data.get('maxAmount', 2000)
        && request.resource.data.transactionId is string
        && request.resource.data.transactionId.size() >= 8
        && reqId.matches('dep_.*')
        && reqId == 'dep_' + request.resource.data.transactionId
        && request.resource.data.requestId == reqId
        && !('reviewedAt' in request.resource.data)
        && !('reviewedBy' in request.resource.data)
        && !('rejectionReason' in request.resource.data)
        && request.resource.data.keys().hasOnly(['userId', 'userName', 'ffUid', 'amount', 'transactionId', 'status', 'createdAt', 'requestId']);
      // Approve/reject: ADMIN ONLY (deliberately NOT staff — deposits
      // credit money; staff is decrease-only).
      allow update, delete: if isAdmin();
    }

    // ── Deposit pending-cap counter (v16, mirrors withdrawalCounters) ──
    // deposit.service.js keeps the user's deposit request IDs here and
    // re-reads them INSIDE the create transaction, so the max-3-pending
    // cap holds even under parallel submits. Only the owner writes it;
    // admin can read for support (v19 M6: staff removed).
    match /depositCounters/{uid} {
      allow read: if isOwner(uid) || isAdmin();
      allow create, update: if isOwner(uid)
        && request.resource.data.uid == uid
        && request.resource.data.ids is list
        && request.resource.data.ids.size() <= 10;
      allow delete: if isAdmin();
    }

    // ── Withdrawal pending-cap counter (v12, AUD-013) ──
    // withdrawal.service.js keeps the user's withdrawal request IDs here
    // and re-reads them INSIDE the create transaction, so the max-3-pending
    // and total-pending ≤ balance caps hold even under parallel submits.
    // Only the owner writes it; admin can read for support (v19 M6: staff removed).
    match /withdrawalCounters/{uid} {
      allow read: if isOwner(uid) || isAdmin();
      allow create, update: if isOwner(uid)
        && request.resource.data.uid == uid
        && request.resource.data.ids is list
        && request.resource.data.ids.size() <= 10;
      allow delete: if isAdmin();
    }

    // ── Wallet ledger ──
    match /walletTransactions/{txnId} {
      // v18 (L2): resource == null guard — missing doc deny নয়, not-found।
      // v19 (M6): staff read removed.
      allow read: if isAdmin()
        || (isSignedIn() && (resource == null || resource.data.userId == request.auth.uid));
      // v11 (A1): create শুধু admin। User app কখনো ledger entry
      // লেখে না (grep যাচাই — wallet.service.js শুধু নিজের history পড়ে);
      // আগে যেকোনো signed-in user নিজের নামে ভুয়া ledger বানাতে পারত।
      // আসল লেখক: admin approve (withdraw_<id>), prize (results.js),
      // manual edit (users.js) — সবাই admin (v19 M6: staff removed)।
      allow create: if isAdmin()
        // v17 (INSTANT SLOT BOOKING): owner শুধু নিজের entry-fee DEBIT
        // ledger লিখতে পারবে — deterministic ID `debit_<teamId>`,
        // amount negative, paymentMethod 'wallet', status 'completed',
        // fixed key list। Credit / prize / refund entry owner লিখতে
        // পারবে না (type 'debit'-ই বাঁধা)। ভুয়া debit লেখা self-harm
        // ছাড়া কিছু নয় — balance বাড়ে না, শুধু history-তে −₹ দেখায়;
        // আসল debit txn-এর ভেতরের user-doc update-ই করে, আর সেটা
        // decrease-only (users rule, উপরে দেখো)।
        || (isSignedIn() && isNotBanned()
            && request.resource.data.userId == request.auth.uid
            && request.resource.data.type == 'debit'
            && request.resource.data.amount is number
            && request.resource.data.amount < 0
            && request.resource.data.paymentMethod == 'wallet'
            && request.resource.data.status == 'completed'
            && txnId == 'debit_' + request.resource.data.teamId
            && request.resource.data.reference == 'WALLET-' + request.resource.data.teamId
            && request.resource.data.keys().hasOnly(['userId', 'type', 'title', 'description', 'paymentMethod', 'reference', 'source', 'tournamentId', 'teamId', 'amount', 'status', 'createdAt']));
      // v16 (wallet): deposit CREDIT entries (idempotent key
      // deposit_<requestId>) and ADMIN wallet-pay DEBIT entries (debit_
      // <paymentId>) are written by the ADMIN confirm txn (admin-panel
      // code). Staff must NOT write deposit credits.
      // v17 (INSTANT SLOT BOOKING): উপরের owner debit branch যোগ হলো —
      // instant join-এর `debit_<teamId>` entry user নিজে লেখে (txn-এর
      // ভেতরে); admin confirm-এর `debit_<paymentId>` আগের মতোই
      // admin-only branch-এ চলে।
      // v14 (W5): update/delete admin-only — staff must never rewrite or
      // erase a ledger entry after the fact (audit trail).
      allow update, delete: if isAdmin();
    }

    // ── Staff activity log ──
    // v19 (M6): staff read/create REMOVED — no staff panel, and no code
    // in either tree reads or writes this collection (write-paths.md).
    // Kept admin-only so past logs stay readable; nothing writes them now.
    match /staffActivityLogs/{logId} {

    match /adminActivityLogs/{logId} {

    // ── Persistent teams — "My Team" (v7) ──
    // One saved squad roster per user. The team is a TEMPLATE: tournament
    // entries snapshot the roster, so deleting a team never touches
    // tournamentTeams docs. Only the captain can write.
    match /teams/{teamId} {

      // entry only). Reads scoped to the owner so old saved rosters are
      // not enumerable by other accounts.
      // v18 (L2): resource == null guard — missing doc deny নয়, not-found।
      allow read: if isAdmin()
        || (isSignedIn() && (resource == null || resource.data.captainUid == request.auth.uid));
      // FIX (2026-11-19, M6): lock শুধু client-side ছিল — rules-এ কোনো
      // lock check ছিল না, captain console থেকে team edit করতে পারত।
      // My Team-এ edit বলতে কিছু নেই (শুধু delete + নতুন করে বানানো),

        && request.resource.data.captainUid == request.auth.uid
        && request.resource.data.members.size() >= 1 && request.resource.data.members.size() <= 3;

        && resource.data.captainUid == request.auth.uid;

    // ── Player reports (v7) ──
    // Users file reports (no auto-ban — admin reviews). Reporter reads own;

    // ── App cache-bust (v15, dashboard H1) ──
